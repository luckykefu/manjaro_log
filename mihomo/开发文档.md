# 安装
```bash
sudo pacman -S --needed --noconfirm mihomo

```
# 下载配置
```bash
readonly CONFIG=/data/.manjaro/mihomo/config.yaml
readonly BOOTSTRAP_PROXY="http://127.0.0.1:7890"
readonly url=https://api3.nimenshishangdi.cc/dazhutou/36e7ae6fb17ac459e2e27d98967fa12a
curl -x "$BOOTSTRAP_PROXY" -sLfA 'clash.meta' --connect-timeout 10 --max-time 30 --retry 2 -o "$CONFIG" "$url"

```
# TUN 配置
```bash
yq -i -y '.tun = {"enable": true, "stack": "mixed", "dns-hijack": ["any:53"], "auto-route": true, "auto-redir": true}' $CONFIG
```

# 启动
```bash
# TUN（root）
# sudo mihomo ...
# 后台
sudo pkill -q mihomo 2>/dev/null || true
nohup sudo mihomo -d "$(dirname "$CONFIG")" > "${CONFIG}.log" 2>&1 &
```

# 验证启动成功
```bash
readonly port=$(yq -r '.["mixed-port"] // 7890' "$CONFIG")
for (( i=1; i<=5; i++ )); do
    ss -tlnp 2>/dev/null | grep ":${port} " && return 0
    echo "Waiting for :$port ... ($i/5)"
    sleep 1
done
```

# 验证网络连通性 200
```bash
curl -sx "http://127.0.0.1:$port" -o /dev/null -w "%{http_code}" --connect-timeout 10 "http://www.google.com" 2>/dev/null || true
```

# 开放本地端口,允许局域网连接
```bash
sudo pacman -S --needed --noconfirm ufw
sudo ufw enable
sudo ufw allow "$port/tcp" 2>/dev/null || true

```
